# =============
# == SOURCES ==
# =============

set(PUBLIC_INCLUDE_FILES
  include/onbings-vista/vista.h
  include/onbings-vista/vista_engine.h
)

set(SRC_FILES
  src/vista_engine.cpp
  src/vista_web_server.cpp
  src/vista_web_server_business.cpp  
)

set(CURRENT_LIB_NAME onbings_vista)
configure_file(
  ${PROJECT_SOURCE_DIR}/compiled_version.h.in
  ${CMAKE_CURRENT_BINARY_DIR}/onbings-vista/compiled_version.h
  @ONLY
)

set(PUBLIC_GENERATED_INCLUDE_FILES
  ${CMAKE_CURRENT_BINARY_DIR}/onbings-vista/compiled_version.h
)

# Logical groups
source_group(TREE ${CMAKE_CURRENT_SOURCE_DIR}/include PREFIX "public" FILES ${PUBLIC_INCLUDE_FILES})
source_group(TREE ${CMAKE_CURRENT_SOURCE_DIR}/src PREFIX "src" FILES ${SRC_FILES})

# Create the assets and binaries/lib directories in the build tree BEFORE configure_file
# This ensures these are proper directories, not files created by configure_file
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/binaries/bin/assets/www")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/binaries/lib")

find_package(Python REQUIRED COMPONENTS Interpreter)

#
# In /home/bha/pro/onbings-vista/code/sources/onbings-vista-service/assets use
# pip install PyYAML
# python3 swagger-yaml-to-html.py --input ./rest-vista.yaml --output rest-vista.html
# to generate rest-vista.html
# Deploy the assets directory
# If you go to the /doc route you will have a test web page to test the api
# 
execute_process(
  COMMAND "${Python_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/assets/swagger-yaml-to-html.py"
  --input "${CMAKE_CURRENT_SOURCE_DIR}/assets/rest-vista.yaml"
  --output "${CMAKE_BINARY_DIR}/binaries/bin/assets/www/rest-vista.html"
  WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/assets"
  RESULT_VARIABLE _help_html_result
  OUTPUT_VARIABLE _help_html_out
  ERROR_VARIABLE _help_html_err
)
message(STATUS "Swagger YAML to HTML stdout: ${_help_html_out}")
message(STATUS "Swagger YAML to HTML stderr: ${_help_html_err}")
if(NOT _help_html_result EQUAL 0)
  message(FATAL_ERROR "Failed to generate rest-vista.html: ${_help_html_err}")
endif()

# Copy all contents from the source assets directory to the build assets directory
message("Copy assets files from ${CMAKE_CURRENT_SOURCE_DIR}/assets to ${CMAKE_BINARY_DIR}/")
file(COPY "${CMAKE_CURRENT_SOURCE_DIR}/assets" DESTINATION "${CMAKE_BINARY_DIR}/binaries/bin")
# ===========
# == BUILD ==
# ===========

add_library(onbings-vista-objects OBJECT
  ${PUBLIC_INCLUDE_FILES}
  ${SRC_FILES}
  ${PUBLIC_GENERATED_INCLUDE_FILES}
)

# Create alias
add_library(ONBINGS::onbings-vista-objects ALIAS onbings-vista-objects)

# Include
target_include_directories(onbings-vista-objects
  PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>

  PRIVATE
)

# Link
target_link_libraries(onbings-vista-objects
  PUBLIC
  PRIVATE
  EVS::evs-hwfw-ilf
  ONBINGS::bofstd
)
# Defines
target_compile_definitions(onbings-vista-objects
  PUBLIC

  PRIVATE
  # Make directives export
  -Donbings_vista_EXPORTS=1
)

if(MSVC)
  target_compile_options(onbings-vista-objects PRIVATE /arch:AVX2)
else()
  target_compile_options(onbings-vista-objects PRIVATE -mavx2 -mfma)
endif()

# Feature
target_compile_features(onbings-vista-objects
  PUBLIC

  PRIVATE
)
# ====================
# == CREATE LIBRARY ==
# ====================

# Now create the "real" library
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/Dummy.cpp)

# Make sure we have at least one file
add_library(onbings-vista
  ${CMAKE_CURRENT_BINARY_DIR}/Dummy.cpp
)

# Link with object library
target_link_libraries(onbings-vista
  PUBLIC
  ONBINGS::onbings-vista-objects
)

# Create alias
add_library(ONBINGS::onbings-vista ALIAS onbings-vista)

# ====================
# == API management ==
# ====================

include(GenerateExportHeader)

# This CMake function generates a header file (i.e. onbings-vista_export.h)
# that contains all the proper 'defines' to import/export symbols
# (e.g. ONBINGS_VISTA_EXPORT) for shared libraries. When building as static
# those macros are simply blank. This file will be by default generated
# into ${CMAKE_CURRENT_BINARY_DIR} (see target_include_directories above)
generate_export_header(onbings-vista)

# =============
# == INSTALL ==
# =============

if(NOT APPLE)

  # Fix RPATH upon installation by replacing
  # the absolute RPATH generated by the compiler
  # by a keyword that basically says 'look for my
  # dependencies in the directory I'm currently
  # installed'.
  set_target_properties(onbings-vista PROPERTIES
    INSTALL_RPATH $ORIGIN/../lib
    # Same as service: keep sysroot paths for AppImage, add $ORIGIN/../lib
    # so the deployed .so finds its dependencies when copied to target.
    BUILD_RPATH "$ORIGIN/../lib"
  )

endif()

# Install the target runtime
install(
  TARGETS onbings-vista onbings-vista-objects
  EXPORT ONBINGS_VISTA
  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} # For binaries and dll
  COMPONENT onbings-vista-runtime
  LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} # For shared libraries
  COMPONENT onbings-vista-runtime
  NAMELINK_COMPONENT onbings-vista-devel
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} # For static libraries
  COMPONENT onbings-vista-devel
)

# Install the public headers
install(FILES ${PUBLIC_INCLUDE_FILES} ${PUBLIC_GENERATED_INCLUDE_FILES}
  DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/onbings-vista
  COMPONENT onbings-vista-devel
)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/onbings-vista_export.h
  DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}
  COMPONENT onbings-vista-devel
)
